#### 1. 课程模块：课程列表支持校长/管理员下拉选择班级

**文件**：`src/pages/Course/CourseListPage.jsx`
**修改**：校长/管理员需要一个下拉框来选择班级才能查到课程。由于现有代码只读 `activeId`，我们需要在顶部的 `headerRight` 中注入一个班级选择器。

```jsx
// 1. 在 CourseListPage 组件顶部引入需要的 API
import { classesApi, schoolApi } from "../../services/api";

// 2. 在 state 中增加：
const [adminClassList, setAdminClassList] = useState([]);
const [selectedAdminClassId, setSelectedAdminClassId] = useState("");

// 3. 增加 useEffect 拉取班级 (管理员/校长逻辑)
useEffect(() => {
  if (isAdmin || isPrincipal) {
    const schoolId = detailInfo?.schoolMembers?.[0]?.schoolId; // 校长默认取管理的第一所学校
    if (schoolId || isAdmin) {
      classesApi
        .get("/all", isAdmin ? {} : { params: { schoolId } })
        .then((res) => {
          if (res.data?.code === 200 && res.data.data.length > 0) {
            setAdminClassList(res.data.data);
            setSelectedAdminClassId(res.data.data[0].id);
          }
        });
    }
  }
}, [isAdmin, isPrincipal, detailInfo]);

// 4. 修改 loadCourses 方法，如果选了班级，则按班级查：
const loadCourses = async (force = false) => {
  setLoading(true);
  try {
    if (isAdmin || isPrincipal) {
      if (!selectedAdminClassId) return; // 还没选班级则不查
      const res = await courseApi.get(`/class/${selectedAdminClassId}`);
      useAuthStore.getState().setCourseList(res.data.data || []); // 手动更新给Store
    } else {
      await fetchCourseList(force);
    }
    setCurrentPage(1);
  } finally {
    setLoading(false);
  }
};

// 监听下拉框改变
useEffect(() => {
  if (selectedAdminClassId) loadCourses(true);
}, [selectedAdminClassId]);

// 5. 在 JSX 的 headerRight 区域加入下拉框：
<div className={styles.headerRight}>
  {(isAdmin || isPrincipal) && (
    <select
      className={styles.pageSizeSelect}
      style={{ marginRight: "10px" }}
      value={selectedAdminClassId}
      onChange={(e) => setSelectedAdminClassId(e.target.value)}
    >
      {adminClassList.map((c) => (
        <option key={c.id} value={c.id}>
          {c.name}
        </option>
      ))}
    </select>
  )}
  {/* 原有的搜索框等... */}
</div>;
```

#### 4. 提交列表多条件查询过滤 (作业状态、分数区间、时间等)

为了实现纯前端的过滤（和目前 Dashboard 逻辑一致），我们需要更新组件内的 `useMemo` 以及渲染搜索框。

**在 `SubmissionList.jsx` 中增加过滤状态及搜索框（影响校长、老师、管理员）：**

```jsx
// 1. 引入 FontAwesome 和状态
import { faFilter } from "@fortawesome/free-solid-svg-icons";
const [filters, setFilters] = useState({
  status: "",
  scoreRange: "",
  date: "",
});

// 2. 过滤逻辑：
const filteredSubmissions = useMemo(() => {
  return submissions.filter((sub) => {
    // 状态
    const matchStatus =
      !filters.status ||
      sub.status === filters.status ||
      (filters.status === "SUBMITTED" && sub.status === "已提交");
    // 分数范围 (例如输入 "80-100")
    let matchScore = true;
    if (filters.scoreRange && sub.score !== null) {
      const [min, max] = filters.scoreRange.split("-").map(Number);
      if (!isNaN(min) && !isNaN(max))
        matchScore = sub.score >= min && sub.score <= max;
      else matchScore = String(sub.score).includes(filters.scoreRange); // 兼容单数字匹配
    } else if (filters.scoreRange && sub.score === null) {
      matchScore = false;
    }
    // 时间
    const matchDate =
      !filters.date ||
      (sub.submitTime && sub.submitTime.startsWith(filters.date));

    return matchStatus && matchScore && matchDate;
  });
}, [submissions, filters]);

// 3. 在 viewHeader 下方添加工具栏：
<div className={styles.toolbar} style={{ margin: "0 0 20px 0" }}>
  <div className={styles.filterGroup}>
    <FontAwesomeIcon icon={faFilter} className={styles.filterIcon} />
    <select
      value={filters.status}
      onChange={(e) => setFilters({ ...filters, status: e.target.value })}
      className={styles.filterInput}
    >
      <option value="">全部状态</option>
      <option value="UNSUBMITTED">未提交</option>
      <option value="SUBMITTED">已提交</option>
      <option value="GRADED">已批改</option>
    </select>
    <input
      type="text"
      placeholder="分数筛选 (例: 80-100或90)"
      value={filters.scoreRange}
      onChange={(e) => setFilters({ ...filters, scoreRange: e.target.value })}
      className={styles.filterInput}
    />
    <input
      type="date"
      value={filters.date}
      onChange={(e) => setFilters({ ...filters, date: e.target.value })}
      className={styles.filterInput}
    />
  </div>
</div>;

// 4. 将原本渲染 `submissions.map` 改为 `filteredSubmissions.map`
```

**对于 `StudentDashboard.jsx` 中的学生视图补齐条件：**
学生原本缺失“所有作业”的时间，“我的提交”的状态、教师名、时间和区间分数。
只需在 `StudentDashboard.jsx` 找到 `const [filters, setFilters] = useState(...)` 加入缺失字段：

```jsx
const [filters, setFilters] = useState({
  status: "",
  courseId: context === "course" ? contextId : "",
  scoreRange: "",
  creatorName: "",
  date: "", // 【新增】时间过滤
});

// filter 条件中增加时间判断：
const matchDate =
  !filters.date ||
  (item.createTime && item.createTime.startsWith(filters.date));

```
